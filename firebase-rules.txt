// Firestore Rules — pegar en Firebase Console > Firestore Database > Rules
// Cualquier cuenta autenticada dentro de tu proyecto puede leer y escribir
// los movimientos, presupuestos y configuración compartidos.
// SEGURIDAD: como solo tú creas las cuentas manualmente en Authentication
// (nadie puede autoregistrarse desde la app, porque el formulario solo
// tiene "Iniciar sesión"), solo tu familia tendrá acceso a estos datos.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /transactions/{transactionId} {
      allow read, create: if request.auth != null;
      allow update, delete: if request.auth != null;
    }
    match /settings/{settingId} {
      allow read, write: if request.auth != null;
    }
    match /budgets/{budgetId} {
      allow read, write: if request.auth != null;
    }
  }
}
